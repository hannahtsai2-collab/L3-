  <script>
    // Structure Tags Definition
    const TAG_DATA = [
      { id: 'tag-1', text: '提出論點' },
      { id: 'tag-2', text: '舉出論據（正例）' },
      { id: 'tag-3', text: '舉出論據（反例）' },
      { id: 'tag-4', text: '提出方法' },
      { id: 'tag-5', text: '重申論點' }
    ];

    // Web Audio Synthesizer for SFX
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    let audioCtx = null;

    function playSound(type) {
      try {
        if (!audioCtx) audioCtx = new AudioCtx();
        if (audioCtx.state === 'suspended') audioCtx.resume();

        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        const now = audioCtx.currentTime;

        if (type === 'correct') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(523.25, now); // C5
          osc.frequency.exponentialRampToValueAtTime(659.25, now + 0.12); // E5
          osc.frequency.exponentialRampToValueAtTime(783.99, now + 0.25); // G5
          gain.gain.setValueAtTime(0.2, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
          osc.start(now);
          osc.stop(now + 0.35);
        } else if (type === 'error') {
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(220, now);
          osc.frequency.linearRampToValueAtTime(140, now + 0.25);
          gain.gain.setValueAtTime(0.2, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
          osc.start(now);
          osc.stop(now + 0.25);
        }
      } catch (e) {
        // Fallback silently if audio context unavailable
      }
    }

    // Utility: Shuffle Array
    function shuffle(array) {
      let currentIndex = array.length, randomIndex;
      const newArray = [...array];
      while (currentIndex !== 0) {
        randomIndex = Math.floor(Math.random() * currentIndex);
        currentIndex--;
        [newArray[currentIndex], newArray[randomIndex]] = [newArray[randomIndex], newArray[currentIndex]];
      }
      return newArray;
    }

    // Render Structure Tag Cards in Random Order
    const tagPoolEl = document.getElementById('tag-pool');

    function createTagElement(tag) {
      const card = document.createElement('div');
      card.className = 'tag-card bg-teal-600 hover:bg-teal-700 text-white text-xs font-bold px-3 py-2 rounded-lg shadow-sm border border-teal-500 cursor-grab flex items-center justify-center gap-1';
      card.draggable = true;
      card.dataset.tagId = tag.id;
      card.dataset.text = tag.text;
      card.innerHTML = `<span>📌</span> <span>${tag.text}</span>`;

      // Drag Events
      card.addEventListener('dragstart', (e) => {
        e.dataTransfer.setData('text/plain', tag.text);
        e.dataTransfer.setData('tag-id', tag.id);
        card.classList.add('opacity-40');
      });

      card.addEventListener('dragend', () => {
        card.classList.remove('opacity-40');
      });

      // Tap / Click placing
      card.addEventListener('click', (e) => {
        if (card.parentElement.classList.contains('drop-target')) {
          // Click inside target -> return to tag pool
          tagPoolEl.appendChild(card);
          card.onclick = null;
        } else {
          // Click in pool -> place in first available target slot
          const emptyTargets = Array.from(document.querySelectorAll('.drop-target')).filter(t => !t.querySelector('.tag-card'));
          if (emptyTargets.length > 0) {
            placeTagInTarget(card, emptyTargets[0]);
          }
        }
      });

      return card;
    }

    function renderTagPool() {
      tagPoolEl.innerHTML = '';
      const shuffled = shuffle(TAG_DATA);
      shuffled.forEach(tag => {
        tagPoolEl.appendChild(createTagElement(tag));
      });
    }

    // Target Slots Setup
    function setupDropTargets() {
      const targets = document.querySelectorAll('.drop-target');

      targets.forEach(target => {
        target.addEventListener('dragover', (e) => {
          e.preventDefault();
          target.classList.add('drag-over');
        });

        target.addEventListener('dragleave', () => {
          target.classList.remove('drag-over');
        });

        target.addEventListener('drop', (e) => {
          e.preventDefault();
          target.classList.remove('drag-over');
          
          const tagId = e.dataTransfer.getData('tag-id');
          if (tagId) {
            const card = document.querySelector(`[data-tag-id="${tagId}"]`);
            if (card) {
              placeTagInTarget(card, target);
            }
          }
        });
      });
    }

    function placeTagInTarget(card, target) {
      // Return previous card if present
      const existingCard = target.querySelector('.tag-card');
      if (existingCard && existingCard !== card) {
        tagPoolEl.appendChild(existingCard);
      }

      // Hide placeholder text
      const placeholder = target.querySelector('.placeholder-text');
      if (placeholder) placeholder.style.display = 'none';

      // Append card into target
      target.appendChild(card);
      target.classList.add('bg-teal-100/50', 'border-teal-500');
    }

    // Submit Validation Handler
    const submitBtn = document.getElementById('submit-btn');
    const modal = document.getElementById('success-modal');
    const modalCloseBtn = document.getElementById('modal-close-btn');

    submitBtn.addEventListener('click', () => {
      let allCorrect = true;

      // 1. Structure Match Validation
      const blocks = document.querySelectorAll('.paragraph-block');
      blocks.forEach(block => {
        const target = block.querySelector('.drop-target');
        const expectedTag = target.getAttribute('data-target-tag');
        const placedCard = target.querySelector('.tag-card');

        if (!placedCard || placedCard.dataset.text !== expectedTag) {
          allCorrect = false;
          block.classList.add('shake', 'border-red-400', 'bg-red-50/20');
          setTimeout(() => {
            block.classList.remove('shake');
          }, 400);
        } else {
          block.classList.remove('border-red-400', 'bg-red-50/20');
        }
      });

      // 2. Keyword Fill-in Validation
      const inputs = document.querySelectorAll('.key-input');
      inputs.forEach(input => {
        const val = input.value.trim();
        const isFree = input.dataset.free === 'true';

        if (isFree) {
          // Free response non-empty check
          if (val === '') {
            allCorrect = false;
            input.classList.add('border-red-500', 'bg-red-50');
          } else {
            input.classList.remove('border-red-500', 'bg-red-50');
          }
        } else {
          // Exact keyword answer check
          const expected = input.dataset.answer;
          if (val !== expected) {
            allCorrect = false;
            input.classList.add('border-red-500', 'bg-red-50');
          } else {
            input.classList.remove('border-red-500', 'bg-red-50');
            input.classList.add('border-teal-500', 'bg-teal-50/50');
          }
        }
      });

      if (!allCorrect) {
        playSound('error');
      } else {
        playSound('correct');

        // Embed student's answers for Paragraphs 6 & 7 dynamically into Overview Card
